import java.io.*;
import java.util.*;
import java.util.stream.*;

public class Solution {

    public static int[] maxSubarray(int[] arr) {
        // 1. Maximum Subarray Sum (Kadane's Algorithm)
        int maxSubarray = arr[0];
        int currentSum = arr[0];
        
        for (int i = 1; i < arr.length; i++) {
            currentSum = Math.max(arr[i], currentSum + arr[i]);
            maxSubarray = Math.max(maxSubarray, currentSum);
        }
        
        // 2. Maximum Subsequence Sum (Java 8 Streams)
        int maxElement = Arrays.stream(arr).max().orElse(Integer.MIN_VALUE);
        
        int maxSubsequence = maxElement <= 0 
            ? maxElement 
            : Arrays.stream(arr).filter(x -> x > 0).sum();
            
        return new int[]{maxSubarray, maxSubsequence};
    }

    public static void main(String[] args) throws IOException {
        try (BufferedReader br = new BufferedReader(new InputStreamReader(System.in))) {
            String line = br.readLine();
            if (line == null) return;
            
            int t = Integer.parseInt(line.trim());
            StringBuilder sb = new StringBuilder();
            
            while (t-- > 0) {
                br.readLine(); // Skip N as array parsing parses tokens directly
                
                int[] arr = Arrays.stream(br.readLine().trim().split("\\s+"))
                                  .mapToInt(Integer::parseInt)
                                  .toArray();
                
                int[] result = maxSubarray(arr);
                sb.append(result[0]).append(" ").append(result[1]).append("\n");
            }
            System.out.print(sb);
        }
    }
}
