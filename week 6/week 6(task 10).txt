import java.io.*;
import java.util.*;

public class Solution {

    public static int[] circularPalindromes(String s) {
        int n = s.length();
        int[] result = new int[n];
        
        // 1. Construct the doubled string S + S
        String doubled = s + s;
        
        // 2. Preprocess string for Manacher's (e.g., "aba" -> "#a#b#a#")
        StringBuilder sb = new StringBuilder();
        sb.append('^'); // Start boundary marker
        for (int i = 0; i < doubled.length(); i++) {
            sb.append('#').append(doubled.charAt(i));
        }
        sb.append("#$"); // End boundary marker
        String t = sb.toString();
        
        // 3. Execute Manacher's Algorithm
        int m = t.length();
        int[] p = new int[m]; // Store palindrome radii
        int center = 0, right = 0;
        
        for (int i = 1; i < m - 1; i++) {
            int iMirror = 2 * center - i;
            if (right > i) {
                p[i] = Math.min(right - i, p[iMirror]);
            } else {
                p[i] = 0;
            }
            
            // Expand around center i
            while (t.charAt(i + 1 + p[i]) == t.charAt(i - 1 - p[i])) {
                p[i]++;
            }
            
            // Update center and right boundary window
            if (i + p[i] > right) {
                center = i;
                right = i + p[i];
            }
        }
        
        // 4. Calculate the max palindrome length for each rotation window
        for (int i = 0; i < n; i++) {
            int maxLen = 0;
            // The valid window boundaries in the transformed string t
            int startT = 2 * i + 1;
            int endT = 2 * (i + n) + 1;
            
            // Scan through all valid centers that could reside within the rotation window
            for (int j = startT; j <= endT; j++) {
                // Maximum radius bounded by the edges of the current rotation window
                int maxPossibleRadius = Math.min(j - startT, endT - j);
                int actualRadius = Math.min(p[j], maxPossibleRadius);
                
                if (actualRadius > maxLen) {
                    maxLen = actualRadius;
                }
            }
            result[i] = maxLen;
        }
        
        return result;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String line = br.readLine();
        if (line == null) return;
        
        int n = Integer.parseInt(line.trim());
        String s = br.readLine().trim();
        
        int[] result = circularPalindromes(s);
        
        StringBuilder sb = new StringBuilder();
        for (int val : result) {
            sb.append(val).append("\n");
        }
        System.out.print(sb);
    }
}
