Program:
import java.io.*;

public class Solution {

    public static int marsExploration(String s) {
        int alteredCount = 0;
        int n = s.length();
        
        // Loop through every character in the received signal
        for (int i = 0; i < n; i++) {
            char current_char = s.charAt(i);
            
            // Determine the expected character based on the position inside "SOS"
            if (i % 3 == 0 && current_char != 'S') {
                alteredCount++;
            } else if (i % 3 == 1 && current_char != 'O') {
                alteredCount++;
            } else if (i % 3 == 2 && current_char != 'S') {
                alteredCount++;
            }
        }
        
        return alteredCount;
    }

    public static void main(String[] args) throws IOException {
        try (BufferedReader br = new BufferedReader(new InputStreamReader(System.in))) {
            String s = br.readLine();
            if (s != null) {
                System.out.println(marsExploration(s.trim()));
            }
        }
    }
}
