class Result {

    /*
     * Complete the 'matrixRotation' function below.
     *
     * The function accepts following parameters:
     *  1. 2D_INTEGER_ARRAY matrix
     *  2. INTEGER r
     */

    public static void matrixRotation(List<List<Integer>> matrix, int r) {
        int m = matrix.size();
        int n = matrix.get(0).size();
        int layers = Math.min(m, n) / 2;
        
        // Create a temporary structure to store the updated output matrix
        int[][] result = new int[m][n];

        for (int layer = 0; layer < layers; layer++) {
            List<Integer> perimeter = new ArrayList<>();
            
            // Define the layer boundaries
            int top = layer;
            int bottom = m - 1 - layer;
            int left = layer;
            int right = n - 1 - layer;

            // Step 1: Collect elements in an anti-clockwise sequence
            // Top row (left to right)
            for (int j = left; j <= right; j++) perimeter.add(matrix.get(top).get(j));
            // Right col (top + 1 to bottom)
            for (int i = top + 1; i <= bottom; i++) perimeter.add(matrix.get(i).get(right));
            // Bottom row (right - 1 down to left)
            for (int j = right - 1; j >= left; j--) perimeter.add(matrix.get(bottom).get(j));
            // Left col (bottom - 1 down to top + 1)
            for (int i = bottom - 1; i > top; i--) perimeter.add(matrix.get(i).get(left));

            // Step 2: Use modulo to eliminate full-circle redundant loops
            int perimeterSize = perimeter.size();
            int shift = r % perimeterSize;

            // Step 3: Put elements back into the result matrix at shifted positions
            int index = 0;
            
            // Top row
            for (int j = left; j <= right; j++) {
                result[top][j] = perimeter.get((index + shift) % perimeterSize);
                index++;
            }
            // Right col
            for (int i = top + 1; i <= bottom; i++) {
                result[i][right] = perimeter.get((index + shift) % perimeterSize);
                index++;
            }
            // Bottom row
            for (int j = right - 1; j >= left; j--) {
                result[bottom][j] = perimeter.get((index + shift) % perimeterSize);
                index++;
            }
            // Left col
            for (int i = bottom - 1; i > top; i--) {
                result[i][left] = perimeter.get((index + shift) % perimeterSize);
                index++;
            }
        }

        // Step 4: Print the final matrix as space-separated tokens per line
        for (int i = 0; i < m; i++) {
            StringBuilder rowStr = new StringBuilder();
            for (int j = 0; j < n; j++) {
                rowStr.append(result[i][j]).append(" ");
            }
            System.out.println(rowStr.toString().trim());
        }
    }

}

public class Solution {
    public static void main(String[] args) throws IOException {
        BufferedReader bufferedReader = new BufferedReader(new InputStreamReader(System.in));

        String[] firstMultipleInput = bufferedReader.readLine().replaceAll("\\s+$", "").split(" ");

        int m = Integer.parseInt(firstMultipleInput[0]);

        int n = Integer.parseInt(firstMultipleInput[1]);

        int r = Integer.parseInt(firstMultipleInput[2]);

        List<List<Integer>> matrix = new ArrayList<>();

        IntStream.range(0, m).forEach(i -> {
            try {
                matrix.add(
                    Stream.of(bufferedReader.readLine().replaceAll("\\s+$", "").split(" "))
                        .map(Integer::parseInt)
                        .collect(toList())
                );
            } catch (IOException ex) {
                throw new RuntimeException(ex);
            }
        });

        Result.matrixRotation(matrix, r);

        bufferedReader.close();
    }
}
