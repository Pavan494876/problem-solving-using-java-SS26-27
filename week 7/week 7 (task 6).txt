import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.stream.Collectors;

public class Solution {

    public static List<Integer> gradingStudents(List<Integer> grades) {
        return grades.stream()
            .map(grade -> {
                // Rule 1: No rounding if the grade is less than 38
                if (grade < 38) {
                    return grade;
                }
                
                // Rule 2: Round up if the next multiple of 5 is less than 3 points away
                int remainder = grade % 5;
                if (remainder >= 3) {
                    return grade + (5 - remainder);
                }
                
                return grade;
            })
            .collect(Collectors.toList());
    }

    public static void main(String[] args) throws IOException {
        // Fast I/O handling for HackerRank inputs
        BufferedReader bufferedReader = new BufferedReader(new InputStreamReader(System.in));

        // Read the total number of students
        int gradesCount = Integer.parseInt(bufferedReader.readLine().trim());

        List<Integer> grades = new ArrayList<>();

        // Read each grade line by line
        for (int i = 0; i < gradesCount; i++) {
            int gradesItem = Integer.parseInt(bufferedReader.readLine().trim());
            grades.add(gradesItem);
        }

        // Process the grades using the Java 8 stream function
        List<Integer> result = gradingStudents(grades);

        // Print each rounded grade on a new line to match Expected Output
        result.forEach(System.out::println);

        bufferedReader.close();
    }
}
